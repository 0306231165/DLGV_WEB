-- -------------------------------------------------------------------------

CREATE TABLE `NhomDichVu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `ma_nhom` VARCHAR(50) NOT NULL UNIQUE,
    `ten_nhom` VARCHAR(150) NOT NULL,
    `icon` VARCHAR(500) NULL,
    `thu_tu_hien_thi` INT DEFAULT 0,
    `trang_thai` BOOLEAN DEFAULT TRUE
);

CREATE TABLE `DichVu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `nhom_dich_vu_id` INT NOT NULL,
    `ten_dich_vu` VARCHAR(255) NOT NULL,
    `cap_do_dich_vu` VARCHAR(50) NOT NULL, -- 'CoBan', 'CaoCap'
    `is_noi_bat` BOOLEAN DEFAULT FALSE,
    `co_bien_the` BOOLEAN NOT NULL,
    `don_gia_co_ban` DECIMAL(12,2) NULL,
    `thoi_gian_chuan_co_ban` INT NULL, -- Tính bằng phút
    `mo_ta` VARCHAR(500) NULL,
    `noi_dung_chi_tiet` LONGTEXT NULL,
    `trang_thai` BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (`nhom_dich_vu_id`) REFERENCES `NhomDichVu`(`id`)
);

CREATE TABLE `TuyChonBienTheDichVu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `dich_vu_id` INT NOT NULL,
    `ten_tuy_chon` VARCHAR(150) NOT NULL, 
    `don_vi_tinh` VARCHAR(50) NOT NULL DEFAULT 'Phut',
    `thoi_gian_chuan` INT NOT NULL,
    `don_gia` DECIMAL(12,2) NOT NULL,
    `trang_thai` BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (`dich_vu_id`) REFERENCES `DichVu`(`id`)
);

CREATE TABLE `LoaiGoiDichVu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `ten_loai_goi` VARCHAR(100) NOT NULL, -- 'Ca lẻ', 'Gói tháng', 'Gói 24/7'
    `mo_ta` VARCHAR(255) NULL,
    `trang_thai` BOOLEAN DEFAULT TRUE
);

CREATE TABLE `DichVu_LoaiGoi` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `dich_vu_id` INT NOT NULL,
    `loai_goi_id` INT NOT NULL,
    `trang_thai` BOOLEAN DEFAULT TRUE,
    UNIQUE KEY `unique_dichvu_loaigoi` (`dich_vu_id`, `loai_goi_id`),
    FOREIGN KEY (`dich_vu_id`) REFERENCES `DichVu`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`loai_goi_id`) REFERENCES `LoaiGoiDichVu`(`id`) ON DELETE CASCADE
);

CREATE TABLE `CauHinhGoiThang` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `so_thang` INT NOT NULL,
    `phan_tram_giam` INT NOT NULL,
    `trang_thai` BOOLEAN DEFAULT TRUE
);

CREATE TABLE `KhuyenMai` (

    `ma_code` VARCHAR(50) NOT NULL UNIQUE,
    `tag_hien_thi` VARCHAR(100) NULL,
    `tieu_de` VARCHAR(255) NOT NULL,
    `mo_ta` TEXT NULL,
    `dich_vu_id_ap_dung` INT NULL, 
    `gia_tri_don_toi_thieu` DECIMAL(12,2) NOT NULL DEFAULT 0.00,

CREATE TABLE `DichVuThem` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `ten_dv_them` VARCHAR(150) NOT NULL UNIQUE,
    `icon` VARCHAR(255) NULL,
    `mo_ta` VARCHAR(255) NULL,
    `trang_thai` BOOLEAN DEFAULT TRUE
);

CREATE TABLE `DichVu_DichVuThem` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `dich_vu_id` INT NOT NULL,
    `dich_vu_them_id` INT NOT NULL,
    `gia_cong_them` DECIMAL(12,2) NOT NULL,

-- -------------------------------------------------------------------------
-- MODULE 2: HỆ THỐNG TÀI KHOẢN KẾ THỪA CHUNGS (GỘP THEO Ý CÔ)

-- Bảng gốc chứa thông tin định danh cốt lõi
CREATE TABLE `TaiKhoan` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `so_dien_thoai` VARCHAR(15) NOT NULL UNIQUE,
    `mat_khau` VARCHAR(255) NOT NULL,
    `ho_ten` VARCHAR(150) NOT NULL,

    `trang_thai` VARCHAR(50) DEFAULT 'HoatDong'
);

-- Vai trò Khách Hàng kế thừa từ TaiKhoan

    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `tai_khoan_id` INT NOT NULL UNIQUE,
    FOREIGN KEY (`tai_khoan_id`) REFERENCES `TaiKhoan`(`id`) ON DELETE CASCADE
);

-- Vai trò Nhân Viên kế thừa từ TaiKhoan (Bổ sung chỉ số kinh nghiệm & Đã bỏ tọa độ)
CREATE TABLE `NhanVien` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `tai_khoan_id` INT NOT NULL UNIQUE,
    `avatar` VARCHAR(255) NULL,
    `cccd` VARCHAR(20) NOT NULL UNIQUE,
    `ngay_sinh` DATE NOT NULL,
    `gioi_tinh` VARCHAR(10) NOT NULL,
    `dia_chi` VARCHAR(255) NOT NULL,

    `tong_so_danh_gia` INT DEFAULT 0,
    `tong_so_ca_hoan_thanh` INT DEFAULT 0,
    `tong_gio_lam` INT DEFAULT 0,
    
    FOREIGN KEY (`tai_khoan_id`) REFERENCES `TaiKhoan`(`id`) ON DELETE CASCADE
);

-- Đã lược bỏ kinh_do và vi_do
CREATE TABLE `DiaChiDaLuu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `khach_hang_id` INT NOT NULL,
    `ten_goi_nho` VARCHAR(100) NOT NULL, 
    `dia_chi_chi_tiet` VARCHAR(255) NOT NULL,
    FOREIGN KEY (`khach_hang_id`) REFERENCES `KhachHang`(`id`) ON DELETE CASCADE

CREATE TABLE `LienHeDaLuu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `khach_hang_id` INT NOT NULL,
    `ten_nguoi_nhan` VARCHAR(150) NOT NULL,
    `sdt_nhan` VARCHAR(15) NOT NULL,
    FOREIGN KEY (`khach_hang_id`) REFERENCES `KhachHang`(`id`) ON DELETE CASCADE
);

CREATE TABLE `NhanVienYeuThich` (
    `khach_hang_id` INT NOT NULL,
    `nhan_vien_id` INT NOT NULL,
    PRIMARY KEY (`khach_hang_id`, `nhan_vien_id`),
    FOREIGN KEY (`khach_hang_id`) REFERENCES `KhachHang`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`nhan_vien_id`) REFERENCES `NhanVien`(`id`) ON DELETE CASCADE
);

CREATE TABLE `NhanVien_DichVu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `nhan_vien_id` INT NOT NULL,
    `dich_vu_id` INT NOT NULL,
    `trang_thai_duyet` VARCHAR(50) DEFAULT 'ChoDuyet', 
    `ngay_dang_ky` DATETIME DEFAULT CURRENT_TIMESTAMP,
    `ngay_duyet` DATETIME NULL,
    UNIQUE KEY `unique_nhanvien_dichvu` (`nhan_vien_id`, `dich_vu_id`),
    FOREIGN KEY (`nhan_vien_id`) REFERENCES `NhanVien`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`dich_vu_id`) REFERENCES `DichVu`(`id`) ON DELETE CASCADE
);

    `vi_tien_id` INT NOT NULL,
    `ma_giao_dich` VARCHAR(100) NOT NULL UNIQUE,
