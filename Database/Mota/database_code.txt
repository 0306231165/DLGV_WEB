-- -------------------------------------------------------------------------
-- MODULE 1: CẤU HÌNH DỊCH VỤ & GIÁ (MASTER DATA)
-- -------------------------------------------------------------------------

CREATE TABLE `DichVu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `ten_dich_vu` VARCHAR(255) NOT NULL,
    `loai_ca_le` VARCHAR(50) NOT NULL,       -- BinhThuong, ChuyenSau
    `co_tieu_chi_chi_tiet` BOOLEAN NOT NULL, -- TRUE: Có option diện tích/số phòng..., FALSE: Giá cố định
    
    -- CỘT CỨU NGUY ĐƯỢC THÊM VÀO ĐÂY:
    `don_gia_co_ban` DECIMAL(12,2) NULL,     -- Điền giá vào đây KHI co_tieu_chi_chi_tiet = FALSE

    -- PHẦN HIỂN THỊ NỘI DUNG
    `mo_ta` VARCHAR(500) NULL,               -- MÔ TẢ NGẮN: hiển thị ở trang danh sách bên ngoài
    `noi_dung_chi_tiet` LONGTEXT NULL,
    `trang_thai` BOOLEAN DEFAULT TRUE
);

CREATE TABLE `NhomDichVu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `ma_nhom` VARCHAR(50) NOT NULL UNIQUE, -- Vd: 'family', 'care', 'popular'
    `ten_nhom` VARCHAR(150) NOT NULL,      -- Vd: 'Chăm sóc gia đình', 'Phổ biến'
    `icon` VARCHAR(500) NULL,               -- Vd: 'volunteer_activism' (Tên icon Material)
    `thu_tu_hien_thi` INT DEFAULT 0,       -- Dùng để sắp xếp tab nào đứng trước, tab nào đứng sau
    `trang_thai` BOOLEAN DEFAULT TRUE
);

CREATE TABLE `DichVu_NhomDichVu` (
    `dich_vu_id` INT NOT NULL,
    `nhom_dich_vu_id` INT NOT NULL,
    PRIMARY KEY (`dich_vu_id`, `nhom_dich_vu_id`),
    FOREIGN KEY (`dich_vu_id`) REFERENCES `DichVu`(`id`),
    FOREIGN KEY (`nhom_dich_vu_id`) REFERENCES `NhomDichVu`(`id`)
);

CREATE TABLE `TanSuat` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `ten_tan_suat` VARCHAR(100) NOT NULL,
    `mo_ta` VARCHAR(255) NULL,
    `trang_thai` BOOLEAN DEFAULT TRUE -- [GHI CHÚ SOFT DELETE]: Đã thêm để an toàn dữ liệu
);

CREATE TABLE `DichVu_TanSuat` (
    `dich_vu_id` INT NOT NULL,
    `tan_suat_id` INT NOT NULL,
    PRIMARY KEY (`dich_vu_id`, `tan_suat_id`),
    FOREIGN KEY (`dich_vu_id`) REFERENCES `DichVu`(`id`),
    FOREIGN KEY (`tan_suat_id`) REFERENCES `TanSuat`(`id`)
);

CREATE TABLE `CauHinhGoiThang` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `so_thang` INT NOT NULL,
    `phan_tram_giam` INT NOT NULL,
    `trang_thai` BOOLEAN DEFAULT TRUE
);

CREATE TABLE `DichVuThem` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `dich_vu_id` INT NOT NULL,
    `ten_dv_them` VARCHAR(150) NOT NULL,
    `gia_cong_them` DECIMAL(12,2) NOT NULL,
    `thoi_gian_cong` INT NOT NULL,
    `trang_thai` BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (`dich_vu_id`) REFERENCES `DichVu`(`id`)
);

CREATE TABLE `TieuChiChiTiet` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `dich_vu_id` INT NOT NULL,
    `ten_tieu_chi` VARCHAR(150) NOT NULL,
    `trang_thai` BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (`dich_vu_id`) REFERENCES `DichVu`(`id`)
);

CREATE TABLE `OptionTieuChi` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `tieu_chi_id` INT NOT NULL,
    `ten_option` VARCHAR(150) NOT NULL,
    
    -- Khai báo đơn vị tính để App biết hiển thị
    `don_vi_tinh` VARCHAR(50) NOT NULL DEFAULT 'Cai', -- Vd: 'Cai', 'Bo', 'm2', 'Goi/Ca(Phút)'
    
    `thoi_gian_chuan` INT NOT NULL,                   -- Số phút làm cho 1 đơn vị
    `don_gia` DECIMAL(12,2) NOT NULL,                 -- Giá tiền cho 1 đơn vị
    `trang_thai` BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (`tieu_chi_id`) REFERENCES `TieuChiChiTiet`(`id`)
);

-- -------------------------------------------------------------------------
-- MODULE 2: NGƯỜI DÙNG, NHÂN VIÊN & TÀI KHOẢN

CREATE TABLE `KhachHang` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `ho_ten` VARCHAR(150) NOT NULL,
    `so_dien_thoai` VARCHAR(15) NOT NULL UNIQUE,
    `mat_khau` VARCHAR(255) NOT NULL,
    `email` VARCHAR(100) NULL,
    `ngay_tao` DATETIME DEFAULT CURRENT_TIMESTAMP,
    `trang_thai` VARCHAR(50) NOT NULL
);

CREATE TABLE `DiaChiDaLuu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `khach_hang_id` INT NOT NULL,
    `ten_goi_nho` VARCHAR(100) NOT NULL,
    `dia_chi_chi_tiet` VARCHAR(255) NOT NULL,
    `kinh_do` DECIMAL(10,7) NULL,
    `vi_do` DECIMAL(10,7) NULL,
    FOREIGN KEY (`khach_hang_id`) REFERENCES `KhachHang`(`id`)
);

CREATE TABLE `LienHeDaLuu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `khach_hang_id` INT NOT NULL,
    `ten_nguoi_nhan` VARCHAR(150) NOT NULL,
    `sdt_nhan` VARCHAR(15) NOT NULL,
    FOREIGN KEY (`khach_hang_id`) REFERENCES `KhachHang`(`id`)
);

CREATE TABLE `NhanVien` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `ho_ten` VARCHAR(150) NOT NULL,
    `so_dien_thoai` VARCHAR(15) NOT NULL UNIQUE,
    `mat_khau` VARCHAR(255) NOT NULL,
    `avatar` VARCHAR(255) NULL,
    `cccd` VARCHAR(20) NOT NULL UNIQUE,
    `ngay_sinh` DATE NOT NULL,
    `gioi_tinh` VARCHAR(10) NOT NULL,

    -- [GHI CHÚ GPS]: Định vị tâm điểm hoạt động của nhân viên để lọc khoảng cách (VD: chỉ gợi ý đơn < 5km)
    `kinh_do_hoat_dong` DECIMAL(10,7) NULL,
    `vi_do_hoat_dong` DECIMAL(10,7) NULL,

    `danh_gia_sao` DECIMAL(3,2) DEFAULT 5.00,
    `trang_thai` VARCHAR(50) NOT NULL
);

-- [GHI CHÚ ĐỘT PHÁ]: Bảng lưu lịch đăng ký rảnh/bận 7 ngày trong tuần.
CREATE TABLE `LichCamKet` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `nhan_vien_id` INT NOT NULL,
    `ngay_bat_dau_cam_ket` DATE NOT NULL,
    `ngay_ket_thuc_cam_ket` DATE NOT NULL,
    `thu_trong_tuan` INT NOT NULL,                     -- Từ 2 đến 8
    `ca_dang_ky` VARCHAR(50) NOT NULL,                 -- 'Sang', 'Chieu', 'Toi', 'CaNgay'
    
    -- [GHI CHÚ TRẠNG THÁI]:
    -- 'DangHoatDong' : Chu kỳ đang trong hiệu lực, dùng để ghép lịch
    -- 'HetHan'       : Hệ thống tự set khi ngay_ket_thuc_cam_ket < NOW()
    -- 'DaHuy'        : NV chủ động hủy giữa chừng (nghỉ việc, bỏ app...)
    `trang_thai` VARCHAR(50) DEFAULT 'DangHoatDong',
    
    FOREIGN KEY (`nhan_vien_id`) REFERENCES `NhanVien`(`id`)
);

CREATE TABLE `NhanVienYeuThich` (
    `khach_hang_id` INT NOT NULL,
    `nhan_vien_id` INT NOT NULL,
    PRIMARY KEY (`khach_hang_id`, `nhan_vien_id`),
    FOREIGN KEY (`khach_hang_id`) REFERENCES `KhachHang`(`id`),
    FOREIGN KEY (`nhan_vien_id`) REFERENCES `NhanVien`(`id`)
);

CREATE TABLE `NhanVien_DichVu` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `nhan_vien_id` INT NOT NULL,
    `dich_vu_id` INT NOT NULL,
    `trang_thai_duyet` VARCHAR(50) DEFAULT 'ChoDuyet', -- ChoDuyet, DaDuyet, TuChoi
    `ngay_dang_ky` DATETIME DEFAULT CURRENT_TIMESTAMP,
    `ngay_duyet` DATETIME NULL,
    UNIQUE KEY `unique_nhanvien_dichvu` (`nhan_vien_id`, `dich_vu_id`),
    FOREIGN KEY (`nhan_vien_id`) REFERENCES `NhanVien`(`id`),
    FOREIGN KEY (`dich_vu_id`) REFERENCES `DichVu`(`id`)
);

CREATE TABLE `DonHang` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `khach_hang_id` INT NOT NULL,
    `dich_vu_id` INT NOT NULL,
    `khuyen_mai_id` INT NULL,                         -- Liên kết Mã khuyến mãi
    `tan_suat_id` INT NOT NULL,

    -- [GHI CHÚ ĐẶT ĐÍCH DANH]: Lưu ID nếu khách chọn đích danh từ mục Yêu thích.
    `nhan_vien_duoc_yeu_cau_id` INT NULL,              
    `phuong_an_thay_the` VARCHAR(100) NULL,           -- 'TimPremiumKhac', 'TimTieuChuan', 'HuyDon'
    `is_giu_nhan_vien` BOOLEAN DEFAULT FALSE,         -- Bật TRUE để giữ cố định 1 người suốt gói tháng
    
    -- [GHI CHÚ PHẠT HỦY]: Dành cho việc HỦY TOÀN BỘ HỢP ĐỒNG (VD: Bỏ ngang gói 6 tháng)
    `phi_phat_huy_don` DECIMAL(12,2) DEFAULT 0.00,
    
    -- CẤU HÌNH LỊCH TRÌNH / GÓI THÁNG / 24/7
    `tong_so_buoi` INT NOT NULL DEFAULT 1,            -- Tổng số buổi thực tế của hợp đồng (Vd: 30 buổi)
    `is_lap_lai_hang_tuan` BOOLEAN NOT NULL,
    `cac_ngay_trong_tuan` VARCHAR(50) NULL,           -- Vd: '2,4,6'
    `gio_lam_mac_dinh` TIME NULL,                     -- Giờ làm chuẩn của gói
    `so_thang_goi_thang` INT NULL,
    `ca_lam_247` VARCHAR(50) NULL,                    -- Vd: 'Ngay', 'Dem', 'CaNgay'
    `so_ngay_goi_247` INT NULL,                       
    `tld_giam_goi_thang` INT DEFAULT 0,
    `ngay_bat_dau` DATE NOT NULL,
    `ngay_ket_thuc` DATE NOT NULL,
    
    -- THÔNG TIN TRIỂN KHAI & GHI CHÚ KHÁCH HÀNG
    `ho_ten_thuc_te` VARCHAR(150) NOT NULL,
    `sdt_thuc_te` VARCHAR(15) NOT NULL,
    `dia_chi_thuc_te` VARCHAR(255) NOT NULL,

    -- [GHI CHÚ GPS]: Chốt tọa độ thực tế của nhà khách để map tính khoảng cách tự động
    `kinh_do` DECIMAL(10,7) NULL,
    `vi_do` DECIMAL(10,7) NULL,

    `ghi_chu_cho_nhan_vien` TEXT NULL,                
    
    -- CÁC LỰA CHỌN THÊM (CHECKBOX) & PHỤ PHÍ SNAPSHOT
    `is_cao_cap` BOOLEAN DEFAULT FALSE,               -- TRUE: Nếu khách chọn gói dịch vụ cao cấp
    `phu_phi_cao_cap` DECIMAL(12,2) DEFAULT 0.00,     -- Lưu tổng số tiền 'dịch vụ cao cấp +25%' của toàn bộ gói lớn(+ tiền hết các buổi ca lẻ của gói tháng, không tính giảm giá ưu đãi gói tháng)
    `co_thu_cung` BOOLEAN DEFAULT FALSE,
    `uu_tien_nv_yt` BOOLEAN DEFAULT FALSE,            -- Ưu tiên NV yêu thích
    
    -- TÀI CHÍNH & THANH TOÁN TOÀN DIỆN
    `tong_tien_ban_dau` DECIMAL(12,2) NOT NULL,       -- Tiền gốc khi chưa tính phụ phí cao cấp / giảm giá
    `phu_phi_dat_gap` DECIMAL(12,2) DEFAULT 0.00,
    `tien_giam_khuyen_mai` DECIMAL(12,2) DEFAULT 0.00, -- Lưu cứng tiền được giảm từ Voucher     
    `tong_tien_cuoi_cung` DECIMAL(12,2) NOT NULL,     -- Tiền cuối cùng phải trả (Đã gồm +25% cao cấp và -Voucher)
    `tien_giam_giu` DECIMAL(12,2) DEFAULT 0.00,

    `phuong_thuc_tt` VARCHAR(50) NOT NULL,            -- TienMat, MoMo, ViWeb
    `trang_thai_thanh_toan` VARCHAR(50) DEFAULT 'ChuaThanhToan',
    `ma_giao_dich_online` VARCHAR(255) NULL,          
    
    -- TRẠNG THÁI HỢP ĐỒNG
    `trang_thai_don` VARCHAR(50) NOT NULL,            -- ChoXacNhan, DangThucHien, HoanThanh...
    `ngay_tao` DATETIME DEFAULT CURRENT_TIMESTAMP,
    
    -- KHÓA NGOẠI
    FOREIGN KEY (`khach_hang_id`) REFERENCES `KhachHang`(`id`),
    FOREIGN KEY (`dich_vu_id`) REFERENCES `DichVu`(`id`),
    FOREIGN KEY (`khuyen_mai_id`) REFERENCES `KhuyenMai`(`id`),
    FOREIGN KEY (`tan_suat_id`) REFERENCES `TanSuat`(`id`),
    FOREIGN KEY (`nhan_vien_duoc_yeu_cau_id`) REFERENCES `NhanVien`(`id`)
);

    `nha_cung_cap_tt` VARCHAR(50) NOT NULL,   -- Cổng thanh toán: 'MoMo', 'VNPAY', 'ZaloPay'...
    
    -- CÁC THÔNG SỐ ĐỐI SOÁT
    `ma_giao_dich_noi_bo` VARCHAR(100) NOT NULL UNIQUE, -- Mã App tự sinh ra gửi sang MoMo (Vd: APP-DH123-L1)

    `tin_nhan_loi` VARCHAR(255) NULL,                      -- Dịch mã lỗi ra tiếng người (Vd: 'Khách hàng hủy giao dịch')

    `du_lieu_tra_ve_json` TEXT NULL,          -- Cục dữ liệu MoMo gửi về (Lưu nguyên bản để check chữ ký)

    `nhan_vien_id` INT NULL,                           -- ID nhân viên nhận ca (NULL nếu chưa có ai nhận)

    `thoi_gian_lam_phut` INT NOT NULL,                 -- Số phút làm việc dự kiến
    `loai_ca` VARCHAR(50) NOT NULL,                    -- CaLe, CaGoiThang, CaGoi247

    -- `DeXuat`   : Ngoài phạm vi cam kết (quá km, gói tháng dài hơn...) → NV có thể Accept/Reject

    `ly_do_huy` VARCHAR(255) NULL,                     -- Lý do xin nghỉ/hủy
    `phi_phat_huy_ca` DECIMAL(12,2) DEFAULT 0.00,      -- Tiền phạt áp dụng riêng cho ca lẻ này (Móc thẳng vào Ví NV/KH)
